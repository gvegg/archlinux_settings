# ============================================================
# noto_menu — белая тема GRUB, 1600x900 = GRUB_GFXMODE
#   - фон: белые волны (bg-waves.png, затемнение 25%; см. backgrounds/README.md в репозитории)
#   - три белые плашки со скруглением по центру
#   - подписи пунктов — PNG с прозрачностью (полный AA), поверх плашек
#   - выбранная плашка — синяя рамка (#007aff)
# Подписи генерит make_labels.py (TTF 24px, grayscale AA). pf2-шрифт нужен
# только для терминала и скрытого текста boot_menu (grub-mkfont не умеет AA).
#
# ГЛАВНЫЕ ЛОВУШКИ GRUB (проверено на практике 2026-10-22):
# 1) Плашка 9-patch рисуется высотой  item_height + 2*radius  (пады = радиус).
#    Поэтому при radius=16 и желаемой плашке 44px:  item_height = 12.
#    Шаг строк = item_height + item_spacing; нужен 44+10 = 54 → spacing = 42.
# 2) Компоненты рисуются в ОБРАТНОМ порядке файла: кто позже в theme.txt,
#    тот НИЖЕ. Значит image-подписи идут ДО boot_menu, иначе плашки их закроют.
# ============================================================

desktop-color: "#ffffff"
# 1600x900 = GRUB_GFXMODE, поэтому картинка ложится 1:1 (без масштабирования).
# Фон: "Abstract-background-white-12.jpg" © Davmormar, CC BY-SA 4.0,
# Wikimedia Commons. Изменения: кроп 16:9, ресайз 1600x900, затемнение 25%.
desktop-image: "bg-waves.png"

# Пустая строка-заголовок = шапки сверху нет
title-text: " "

# Терминал (режим правки/boot консоль)
# ВАЖНО (2026-10-23): terminal-border ОБЯЗАТЕЛЬНО 0. При выборе пункта GRUB
# вызывает grub_cls(), который чистит только текстовое поле gfxterm, а полосу
# terminal-border по краям экрана НЕ перерисовывает — на белой теме остаётся
# белая рамка, которая висит и во время загрузки ОС. С border=0 текстовое
# поле занимает весь экран, и grub_cls() гасит его целиком в чёрный.
terminal-font: "Noto Sans Regular 24"
terminal-left: "0"
terminal-top: "0"
terminal-width: "100%"
terminal-height: "100%"
terminal-border: "0"

# ------------------------------------------------------------

# каждая картинка обрезана по чернилам и отцентрована по центру плашки.
# Идут ДО boot_menu — компоненты рисуются в обратном порядке файла.

# ------------------------------------------------------------
# Меню (рисуется ПЕРВЫМ = под подписями).
#   блок: left = (1600-184)/2 = 708, top = (900-152)/2 = 374
#   ширина 184 = подпись «Arch Linux» 119px (24px) + поля 32px с каждой стороны
#   видимая плашка = item_height + 2*16 = 44, шаг = 12 + 42 = 54
# Текст boot_menu скрыт (белый на белой плашке) — подписи рисуют image выше.
# ------------------------------------------------------------
+ boot_menu {
  left = 708
  top = 374
  width = 184
  height = 152

  item_font           = "Noto Sans Regular 24"
  selected_item_font  = "Noto Sans Regular 24"
  item_color          = "#ffffff"   # текст меню скрыт: белым по белой плашке
  selected_item_color = "#ffffff"

  # белые плашки-пункты (9-segment PNG, радиус 16, рамка 2px)
  item_pixmap_style          = "plate_*.png"
  selected_item_pixmap_style = "plate_hi_*.png"

  item_height   = 12    # НЕ высота плашки! плашка = 12 + 2*16 = 44
  item_spacing  = 42    # шаг 12+42 = 54 = 44 плашка + 10 зазор
  item_padding  = 0
  icon_width    = 0
  icon_height   = 0
}
